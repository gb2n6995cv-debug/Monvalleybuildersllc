        // Load saved photo on page load if existing
        window.addEventListener('DOMContentLoaded', function() {
            var savedPhoto = localStorage.getItem('mvb_owner_photo');
            if (savedPhoto) {
                document.getElementById('owner-photo-display').src = savedPhoto;
            }
        });

        // Handle local photo upload from iPhone Camera Roll or File Picker
        function uploadOwnerPhoto(event) {
            var file = event.target.files[0];
            if (file) {
                var reader = new FileReader();
                reader.onload = function(e) {
                    var imgData = e.target.result;
                    document.getElementById('owner-photo-display').src = imgData;
                    try {
                        localStorage.setItem('mvb_owner_photo', imgData);
                    } catch (err) {
                        console.log('Local storage quota reached');
                    }
                };
                reader.readAsDataURL(file);
            }
        }

        // Handle email quote form submission
        function handleDirectEmail(e) {
            e.preventDefault();
            
            var name = document.getElementById('client_name').value;
            var phone = document.getElementById('client_phone').value;
            var service = document.getElementById('client_service').value;
            var notes = document.getElementById('client_notes').value;
            
            var subject = encodeURIComponent("New Estimate Request - " + name);
            var bodyText = "Hello Scott,\n\nI would like to request an estimate for a project:\n\n" +
                           "Name: " + name + "\n" +
                           "Phone: " + phone + "\n" +
                           "Service Needed: " + service + "\n" +
                           "Project Details: " + notes + "\n\n" +
                           "Thank you!";
            
            var mailtoUrl = "mailto:Monvalleybuilders@yahoo.com?subject=" + subject + "&body=" + encodeURIComponent(bodyText);
            
            window.location.href = mailtoUrl;
        }
